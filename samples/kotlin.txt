package demo

data class User(val name: String, val age: Int)

class Stack<T> {
    private val items = mutableListOf<T>()

    fun push(value: T) = apply { items.add(value) }

    fun pop(): T {
        require(items.isNotEmpty()) { "empty stack" }
        return items.removeAt(items.lastIndex)
    }
}

fun main() {
    val names = listOf(User("Ada", 36), User("Alan", 41))
        .filter { it.age >= 18 }
        .map { it.name }
    println(names.joinToString())
    println("found ${names.size} users: ${names.joinToString()}")
}
